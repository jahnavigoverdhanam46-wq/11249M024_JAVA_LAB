import java.applet.Applet;
import java.awt.*;

public class HouseApplet extends Applet {
    public void paint(Graphics g) {
        g.setColor(new Color(240, 190, 120));
        g.fillRect(80, 110, 180, 130);

        g.setColor(Color.RED);
        int[] x = {60, 170, 280};
        int[] y = {110, 35, 110};
        g.fillPolygon(x, y, 3);

        g.setColor(Color.DARK_GRAY);
        g.fillRect(145, 170, 50, 70);

        g.setColor(Color.BLUE);
        g.fillRect(95, 135, 35, 35);
        g.fillRect(210, 135, 35, 35);

        g.setColor(Color.BLACK);
        g.drawRect(80, 110, 180, 130);
    }
}
