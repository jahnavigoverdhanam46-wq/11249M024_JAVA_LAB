interface Printable {
    void print();
}

interface Showable {
    void show();
}

class Demo implements Printable, Showable {
    public void print() {
        System.out.println("Printing details");
    }

    public void show() {
        System.out.println("Showing details");
    }
}

public class InterfaceDemo {
    public static void main(String[] args) {
        Demo d = new Demo();
        d.print();
        d.show();
    }
}
